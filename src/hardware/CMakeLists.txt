# Ring Racers E: Direct3D 9 / 9Ex RHI backend (Windows only).
if(WIN32)
	target_sources(SRB2SDL2 PRIVATE
		../rhi/d3d9/d3d9_rhi.cpp
		../rhi/d3d9/d3d9_shaders.cpp
		../rhi/d3d9/d3dcompiler_embed.cpp
	)

	# Optionally embed d3dcompiler_47.dll inside the executable (Windows 7 does not ship it).
	# Put d3dcompiler_47_x64.dll / d3dcompiler_47_x86.dll (from the Windows SDK redist) in
	# RRE_EMBED_DIR. If the file is missing, nothing is embedded and the game loads the
	# system DLL at runtime instead.
	set(RRE_EMBED_DIR "${CMAKE_SOURCE_DIR}/embed" CACHE PATH "Folder with d3dcompiler_47_x64/x86.dll to embed")
	if(CMAKE_SIZEOF_VOID_P EQUAL 8)
		set(RRE_D3DC_FILE "${RRE_EMBED_DIR}/d3dcompiler_47_x64.dll")
	else()
		set(RRE_D3DC_FILE "${RRE_EMBED_DIR}/d3dcompiler_47_x86.dll")
	endif()
	if(EXISTS "${RRE_D3DC_FILE}")
		message(STATUS "Embedding d3dcompiler: ${RRE_D3DC_FILE}")
		set_source_files_properties(../rhi/d3d9/d3dcompiler_embed.cpp TARGET_DIRECTORY SRB2SDL2 PROPERTIES
			COMPILE_DEFINITIONS "RR_D3DC_PATH=${RRE_D3DC_FILE}" OBJECT_DEPENDS "${RRE_D3DC_FILE}")
	else()
		message(STATUS "d3dcompiler_47 not embedded (not found in ${RRE_EMBED_DIR})")
	endif()

	# Optionally embed ANGLE (DirectX 11 and Vulkan) and Mesa (DirectX 12) inside the executable (64-bit only),
	# so players don't need the angle/ and mesa/ folders. Put the DLLs in RRE_EMBED_DIR/angle and RRE_EMBED_DIR/mesa.
	# They are stored gzip-compressed and written to %LOCALAPPDATA%\RingRacersE\ the first time they are used
	# (rr_gfxpack.cpp). An angle/ or mesa/ folder next to the exe still wins.
	target_sources(SRB2SDL2 PRIVATE rr_gfxpack.cpp)
	set(RRE_PACK_DIR "${CMAKE_CURRENT_BINARY_DIR}/rre_pack")
	set(RRE_PACK_ASM "")
	set(RRE_PACK_TABLE "")
	set(RRE_PACK_DEPS "")
	set(RRE_PACK_N 0)
	if(CMAKE_SIZEOF_VOID_P EQUAL 8)
		foreach(_f angle/libEGL.dll angle/libGLESv2.dll angle/vulkan-1.dll mesa/opengl32.dll mesa/libgallium_wgl.dll mesa/dxil.dll)
			set(_src "${RRE_EMBED_DIR}/${_f}")
			if(EXISTS "${_src}")
				get_filename_component(_g "${_f}" DIRECTORY)
				get_filename_component(_n "${_f}" NAME)
				set(_gz "${RRE_PACK_DIR}/${_g}_${_n}.gz")
				if(NOT EXISTS "${_gz}" OR "${_src}" IS_NEWER_THAN "${_gz}")
					message(STATUS "Compressing ${_f} to embed it")
					file(MAKE_DIRECTORY "${RRE_PACK_DIR}")
					file(ARCHIVE_CREATE OUTPUT "${_gz}" PATHS "${_src}" FORMAT raw COMPRESSION GZip COMPRESSION_LEVEL 9)
				endif()
				string(APPEND RRE_PACK_ASM "__asm__(\".section .rdata,\\\"dr\\\"; .balign 16; .globl rr_pack_${RRE_PACK_N}_s; rr_pack_${RRE_PACK_N}_s: .incbin \\\"${_gz}\\\"; .globl rr_pack_${RRE_PACK_N}_e; rr_pack_${RRE_PACK_N}_e: .text\");\nextern \"C\" const unsigned char rr_pack_${RRE_PACK_N}_s[], rr_pack_${RRE_PACK_N}_e[];\n")
				string(APPEND RRE_PACK_TABLE "\t{\"${_g}\", \"${_n}\", rr_pack_${RRE_PACK_N}_s, rr_pack_${RRE_PACK_N}_e},\n")
				list(APPEND RRE_PACK_DEPS "${_gz}")
				math(EXPR RRE_PACK_N "${RRE_PACK_N} + 1")
			endif()
		endforeach()
	endif()
	if(RRE_PACK_N GREATER 0)
		message(STATUS "Embedding ${RRE_PACK_N} ANGLE/Mesa DLLs from ${RRE_EMBED_DIR}")
		set(_inc "// Generated by src/hardware/CMakeLists.txt\n${RRE_PACK_ASM}static const RRPackFile rr_pack_files[] = {\n${RRE_PACK_TABLE}};\nstatic const int rr_pack_count = ${RRE_PACK_N};\n")
		set(_old "")
		if(EXISTS "${RRE_PACK_DIR}/rre_pack.inc")
			file(READ "${RRE_PACK_DIR}/rre_pack.inc" _old)
		endif()
		if(NOT _old STREQUAL _inc)
			file(WRITE "${RRE_PACK_DIR}/rre_pack.inc" "${_inc}")
		endif()
		target_include_directories(SRB2SDL2 PRIVATE "${RRE_PACK_DIR}")
		set_source_files_properties(rr_gfxpack.cpp TARGET_DIRECTORY SRB2SDL2 PROPERTIES
			COMPILE_DEFINITIONS "RRE_PACK" OBJECT_DEPENDS "${RRE_PACK_DEPS};${RRE_PACK_DIR}/rre_pack.inc")
	endif()
endif()
